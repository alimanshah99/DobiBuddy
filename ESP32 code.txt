String maskPhone(String phone) {
  if (phone.length() <= 6) return phone;
  return phone.substring(0, 3) + "****" + phone.substring(phone.length() - 3);
}

#define BUZZER_PIN 27

int melody[] = {523, 659, 784, 1047};
int duration[] = {200, 200, 200, 400};

void playDoneMelody() {
  for (int i = 0; i < 4; i++) {
    tone(BUZZER_PIN, melody[i]);
    delay(duration[i]);
    noTone(BUZZER_PIN);
    delay(50);
  }
}

void drawDisplay(String status, int displayTimeLeft, String phone) {
  if (status == lastStatus && displayTimeLeft == lastTimeLeft && phone == lastPhone) {
    return;
  }

  lastStatus = status;
  lastTimeLeft = displayTimeLeft;
  lastPhone = phone;

  uint16_t themeColor = ST77XX_WHITE;
  String message = "READY";

  if (status == "RUNNING") {
    themeColor = ST77XX_GREEN;
    message = "WASHING...";
  } 
  else if (status == "PAUSED") {
    themeColor = ST77XX_YELLOW;
    message = "EMERGENCY!";
  } 
  else if (status == "BOOKED") {
    themeColor = ST77XX_BLUE;
    message = "PRESS START";
  } 
  else if (status == "ENDING") {
    themeColor = ST77XX_MAGENTA;
    message = "COLLECT NOW";
  } 
  else if (status == "AVAILABLE") {
    themeColor = ST77XX_CYAN;
    message = "READY";
  }

  int minutes = displayTimeLeft / 60;
  int seconds = displayTimeLeft % 60;

  tft.fillScreen(ST77XX_BLACK);

  // Header
  tft.fillRect(0, 0, 128, 18, themeColor);
  tft.setTextColor(ST77XX_BLACK);
  tft.setTextSize(1);
  tft.setCursor(28, 5);
  tft.print("DobiBuddy");

  // Status box
  tft.drawRoundRect(3, 22, 122, 28, 5, themeColor);
  tft.setTextColor(themeColor);
  tft.setCursor(10, 32);
  tft.print("STATUS: ");
  tft.print(status);

  // Time
  tft.setTextColor(ST77XX_WHITE);
  tft.setCursor(8, 58);
  tft.print("TIME LEFT");

  tft.setTextSize(2);
  tft.setCursor(25, 72);

  if (minutes < 10) tft.print("0");
  tft.print(minutes);
  tft.print(":");
  if (seconds < 10) tft.print("0");
  tft.print(seconds);

  // User
  tft.setTextSize(1);
  tft.setTextColor(ST77XX_WHITE);
  tft.setCursor(8, 96);
  tft.print("USER: ");
  tft.print(maskPhone(phone));

  // Footer message
  tft.fillRoundRect(3, 111, 122, 15, 4, themeColor);
  tft.setTextColor(ST77XX_BLACK);
  tft.setCursor(20, 115);
  tft.print(message);
}

void promoteNextUser() {
  String q0Phone = "", q0Block = "";
  String q1Phone = "", q1Block = "";
  String q2Phone = "", q2Block = "";

  if (Firebase.RTDB.getString(&fbdo, path + "/queueUsers/0/phone")) q0Phone = fbdo.stringData();
  if (Firebase.RTDB.getString(&fbdo, path + "/queueUsers/0/block")) q0Block = fbdo.stringData();

  if (Firebase.RTDB.getString(&fbdo, path + "/queueUsers/1/phone")) q1Phone = fbdo.stringData();
  if (Firebase.RTDB.getString(&fbdo, path + "/queueUsers/1/block")) q1Block = fbdo.stringData();

  if (Firebase.RTDB.getString(&fbdo, path + "/queueUsers/2/phone")) q2Phone = fbdo.stringData();
  if (Firebase.RTDB.getString(&fbdo, path + "/queueUsers/2/block")) q2Block = fbdo.stringData();

  Firebase.RTDB.setBool(&fbdo, path + "/runtime/isRunning", false);
  Firebase.RTDB.setBool(&fbdo, path + "/runtime/isPaused", false);
  Firebase.RTDB.setBool(&fbdo, path + "/notifications/fiveMinSent", false);
  Firebase.RTDB.setBool(&fbdo, path + "/notifications/doneSent", false);
  Firebase.RTDB.setBool(&fbdo, path + "/notifications/nextUserSent", false);

  if (q0Phone != "") {
    Firebase.RTDB.setString(&fbdo, path + "/currentUser/phone", q0Phone);
    Firebase.RTDB.setString(&fbdo, path + "/currentUser/block", q0Block);
    Firebase.RTDB.setInt(&fbdo, path + "/currentUser/timeLeft", 300);

    Firebase.RTDB.setString(&fbdo, path + "/status", "BOOKED");
    Firebase.RTDB.setString(&fbdo, path + "/phase", "PREP");

    if (q1Phone != "") {
      Firebase.RTDB.setString(&fbdo, path + "/queueUsers/0/phone", q1Phone);
      Firebase.RTDB.setString(&fbdo, path + "/queueUsers/0/block", q1Block);
    } else {
      Firebase.RTDB.deleteNode(&fbdo, path + "/queueUsers/0");
    }

    if (q2Phone != "") {
      Firebase.RTDB.setString(&fbdo, path + "/queueUsers/1/phone", q2Phone);
      Firebase.RTDB.setString(&fbdo, path + "/queueUsers/1/block", q2Block);
    } else {
      Firebase.RTDB.deleteNode(&fbdo, path + "/queueUsers/1");
    }

    Firebase.RTDB.deleteNode(&fbdo, path + "/queueUsers/2");

  } else {
    Firebase.RTDB.setString(&fbdo, path + "/currentUser/phone", "");
    Firebase.RTDB.setString(&fbdo, path + "/currentUser/block", "");
    Firebase.RTDB.setInt(&fbdo, path + "/currentUser/timeLeft", 0);

    Firebase.RTDB.setString(&fbdo, path + "/temperature", "");
    Firebase.RTDB.setString(&fbdo, path + "/status", "AVAILABLE");
    Firebase.RTDB.setString(&fbdo, path + "/phase", "IDLE");

    Firebase.RTDB.deleteNode(&fbdo, path + "/queueUsers");
  }
}

void handleControlButton() {
  static bool lastReading = HIGH;
  static unsigned long lastPress = 0;

  bool reading = digitalRead(CONTROL_BTN);

  if (lastReading == HIGH && reading == LOW && millis() - lastPress > 500) {
    lastPress = millis();
    Serial.println("BUTTON PRESSED");

    if (Firebase.RTDB.getString(&fbdo, path + "/status")) {
      String currentStatus = fbdo.stringData();

      if (currentStatus == "BOOKED") {
        Firebase.RTDB.setString(&fbdo, path + "/status", "RUNNING");
        Firebase.RTDB.setString(&fbdo, path + "/phase", "RUNNING");
        Firebase.RTDB.setBool(&fbdo, path + "/runtime/isRunning", true);
        Firebase.RTDB.setBool(&fbdo, path + "/runtime/isPaused", false);
        Firebase.RTDB.setInt(&fbdo, path + "/currentUser/timeLeft", 1800);
      }
      else if (currentStatus == "RUNNING") {
        Firebase.RTDB.setString(&fbdo, path + "/status", "PAUSED");
        Firebase.RTDB.setString(&fbdo, path + "/phase", "PAUSED");
        Firebase.RTDB.setBool(&fbdo, path + "/runtime/isPaused", true);
      }
      else if (currentStatus == "PAUSED") {
        Firebase.RTDB.setString(&fbdo, path + "/status", "RUNNING");
        Firebase.RTDB.setString(&fbdo, path + "/phase", "RUNNING");
        Firebase.RTDB.setBool(&fbdo, path + "/runtime/isPaused", false);
      }
    }
  }

  lastReading = reading;
}

void setup() {
  Serial.begin(115200);

  pinMode(CONTROL_BTN, INPUT_PULLUP);
  pinMode(LED_PIN, OUTPUT);
  //pinMode(TFT_BLK, OUTPUT);

  digitalWrite(LED_PIN, LOW);
  //digitalWrite(TFT_BLK, HIGH);

  SPI.begin(18, -1, 23, TFT_CS);

  tft.initR(INITR_144GREENTAB);
  tft.setRotation(3);
  tft.fillScreen(ST77XX_BLACK);
  tft.setTextColor(ST77XX_GREEN);
  tft.setTextSize(2);
  tft.setCursor(20, 45);
  tft.println("BOOTING");

  initProperties();
  ArduinoCloud.begin(ArduinoIoTPreferredConnection);
  setDebugMessageLevel(2);
  ArduinoCloud.printDebugInfo();

  while (WiFi.status() != WL_CONNECTED) {
    ArduinoCloud.update();
    delay(500);
    Serial.print(".");
  }

  Serial.println();
  Serial.println("WiFi connected by Arduino Cloud");
  Serial.println(WiFi.localIP());

  config.api_key = API_KEY;
  config.database_url = DATABASE_URL;

  if (Firebase.signUp(&config, &auth, "", "")) {
    Serial.println("Firebase OK");
  } else {
    Serial.println("Firebase signup failed");
    Serial.println(config.signer.signupError.message.c_str());
  }

  Firebase.begin(&config, &auth);
  Firebase.reconnectWiFi(true);

  tft.fillScreen(ST77XX_BLACK);
  tft.setTextSize(2);
  tft.setTextColor(ST77XX_CYAN);
  tft.setCursor(20, 35);
  tft.println("DOBI");
  tft.setCursor(15, 65);
  tft.println("BUDDY");
}

void loop() {
  ArduinoCloud.update();
  handleControlButton();

  if (Firebase.ready() && millis() - lastTick >= tickInterval) {
    lastTick = millis();

    if (Firebase.RTDB.getJSON(&fbdo, path)) {
      FirebaseJson json = fbdo.jsonObject();

      FirebaseJsonData statusData;
      FirebaseJsonData timeData;
      FirebaseJsonData phoneData;
      FirebaseJsonData fiveMinData;
      FirebaseJsonData doneData;

      json.get(statusData, "status");
      json.get(timeData, "currentUser/timeLeft");
      json.get(phoneData, "currentUser/phone");
      json.get(fiveMinData, "notifications/fiveMinSent");
      json.get(doneData, "notifications/doneSent");

      String status = statusData.stringValue;
      int machineTimeLeft = timeData.intValue;
      String phone = phoneData.stringValue;
      bool fiveMinSent = fiveMinData.boolValue;
      bool doneSent = doneData.boolValue;

      currentUser = maskPhone(phone);
      machineStatus = status;
      timeLeft = machineTimeLeft;
      machineRunning = (status == "RUNNING");

      drawDisplay(status, machineTimeLeft, phone);

      if (status == "RUNNING") {
        digitalWrite(LED_PIN, HIGH);
      } else {
        digitalWrite(LED_PIN, (millis() / 100) % 2);
      }

      if (status == "BOOKED") {
        if (machineTimeLeft > 0) {
          machineTimeLeft--;
          Firebase.RTDB.setInt(&fbdo, path + "/currentUser/timeLeft", machineTimeLeft);
        }

        if (machineTimeLeft <= 0) {
          promoteNextUser();
        }
      }

      else if (status == "RUNNING") {
        bool isPaused = false;

        if (Firebase.RTDB.getBool(&fbdo, path + "/runtime/isPaused")) {
          isPaused = fbdo.boolData();
        }

        if (!isPaused) {
          if (machineTimeLeft > 0) {
            machineTimeLeft--;
            Firebase.RTDB.setInt(&fbdo, path + "/currentUser/timeLeft", machineTimeLeft);
          }

          if (machineTimeLeft == 300 && !fiveMinSent) {
            Firebase.RTDB.setBool(&fbdo, path + "/notifications/fiveMinSent", true);
          }

          if (machineTimeLeft <= 0 && !doneSent) {
            Firebase.RTDB.setString(&fbdo, path + "/status", "ENDING");
            Firebase.RTDB.setString(&fbdo, path + "/phase", "ENDING");
            Firebase.RTDB.setInt(&fbdo, path + "/currentUser/timeLeft", 300);
            Firebase.RTDB.setBool(&fbdo, path + "/notifications/doneSent", true);
          }
        }
      }

      else if (status == "PAUSED") {
        digitalWrite(LED_PIN, (millis() / 100) % 2);
      }

      else if (status == "ENDING") {
        if (machineTimeLeft > 0) {
          machineTimeLeft--;
          Firebase.RTDB.setInt(&fbdo, path + "/currentUser/timeLeft", machineTimeLeft);
        }

        if (machineTimeLeft <= 0) {
          promoteNextUser();
        }
      }

      else if (status == "AVAILABLE") {
      Firebase.RTDB.setString(&fbdo, path + "/currentUser/phone", "");
      Firebase.RTDB.setString(&fbdo, path + "/currentUser/block", "");
      Firebase.RTDB.setInt(&fbdo, path + "/currentUser/timeLeft", 0);

      Firebase.RTDB.deleteNode(&fbdo, path + "/queueUsers");

      Firebase.RTDB.setString(&fbdo, path + "/phase", "IDLE");
      Firebase.RTDB.setBool(&fbdo, path + "/runtime/isRunning", false);
      Firebase.RTDB.setBool(&fbdo, path + "/runtime/isPaused", false);

      Firebase.RTDB.setBool(&fbdo, path + "/notifications/fiveMinSent", false);
      Firebase.RTDB.setBool(&fbdo, path + "/notifications/doneSent", false);
      Firebase.RTDB.setBool(&fbdo, path + "/notifications/nextUserSent", false);
}

    } else {
      Serial.println("Firebase read failed");
      Serial.println(fbdo.errorReason());
    }
  }
}
